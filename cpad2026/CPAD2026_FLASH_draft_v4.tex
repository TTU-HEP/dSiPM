\begin{frame}{There are two valid studies at two different scales}
  \vspace{-0.03cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      {\large\bfseries One fiber / one sensor}\par
      \vspace{0.14cm}
      \begin{itemize}\tightitem
        \item incident photon density and arrival time;
        \item photons per SPAD as segmentation changes;
        \item PDE, recovery, and local saturation;
        \item timing records and exported payload.
      \end{itemize}
      \vspace{0.22cm}
      \flatbox{CMSBlue}{\bfseries Output: a calibrated channel transfer function.}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\large\bfseries Full calorimeter / full event}\par
      \vspace{0.14cm}
      \begin{itemize}\tightitem
        \item channel-to-channel occupancy and shower cores;
        \item energy linearity and resolution;
        \item transverse imaging and longitudinal timing;
        \item Cherenkov/scintillation response and particle ID.
      \end{itemize}
      \vspace{0.22cm}
      \flatbox{TTURed}{\bfseries Output: bias and information at the physics level.}
    \end{column}
  \end{columns}
  \vspace{0.02cm}
  \centering{\fontsize{9.5}{11}\selectfont\bfseries Close the channel response first; propagate the same model across the detector.}
\end{frame}

    \item Next: digitize current and FLASH readouts from \red{matched full-detector C/S photon samples}, then run an energy scan.

% 20
\begin{frame}{Conclusions}
  \vspace{0.20cm}
  \begin{itemize}\setlength\itemsep{0.54em}
    \item \blue{One fiber is a complete readout study}: occupancy $\rightarrow$ segmentation $\rightarrow$ recovery $\rightarrow$ pulse shape.
    \item The one-fiber result is a joint PDE--pitch--recovery--output operating surface, not one universal pixel size.
    \item \red{HG-DREAM is the detector benchmark}: it tests energy, imaging, timing, and real photon density together.
    \item A 20 $\mu$m, 4.3 ns point preserves the transverse image and reduces the modeled response transfer relative to a representative analog SiPM.
    \item \green{FLASH must be designed against both channel fidelity and calorimeter observables.}
  \end{itemize}
  \vfill
  \centering{\large\bfseries Next milestone: matched full-detector C/S samples $\rightarrow$ energy scan $\rightarrow$ prototype closure.}
\end{frame}

\begin{frame}{Backup: the next simulation set should close the calorimeter loop}
  \begin{enumerate}\tightitem
    \item \red{Generate matched full-detector samples} with explicit Cherenkov and scintillation photons in one configuration.
    \item \red{Run an energy scan} for $e^+$ and $\pi^+$ so linearity and stochastic terms replace single-energy response widths.
